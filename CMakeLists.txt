cmake_minimum_required(VERSION 3.20)
project(learncpp_exercises CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# -pedantic-errors: treat as errors the warnings demanded by strict ISO C/C++ (stricter than -Wpedantic, which only warns)
# -Weffc++: flags violations of Effective C++ style rules (class design, initialization, etc.)
# -Wconversion/-Wsign-conversion: catch implicit narrowing and signed/unsigned conversions
# -Werror: treat all enabled warnings as errors
add_compile_options(-Wall -Weffc++ -Wextra -pedantic-errors -Wconversion -Wsign-conversion -Werror -g)
